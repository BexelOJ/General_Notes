// Rust Projects: (executing a single Rust file)

// Compiling : 
rustc fileName.rs 

// which generates an executable file, named 
fileName

// Running : 
./fileName


------------------------------------------------------------
// Creating a Rust Cargo Project : 

// Create a new binary project:

cargo new project_name --bin


------------------------------------------------------------
// Change into the project directory:

cd project_name

------------------------------------------------------------
// Build the project:

cargo build

------------------------------------------------------------
// Run the project:

cargo run
