# TO make a code run as a daemon service in Ubuntu:
------------------------------
/etc/systemd/system/myapp.service
# service location and name

------------------------------
# You create a systemd service:

[Unit]
Description=My Python Service
After=network.target

[Service]
Type=simple
User=exin
WorkingDirectory=/home/exin/myapp
ExecStart=/home/exin/myapp/venv/bin/python /home/exin/myapp/main.py
Restart=always

[Install]
WantedBy=multi-user.target

------------------------------
# run the service reload and enable cmd

sudo systemctl daemon-reload
sudo systemctl enable myapp
sudo systemctl start myapp
