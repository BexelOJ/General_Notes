#include <iostream>
#include <thread>
#include <string>

using namespace std; 

int add(int a, int b){
    return a + b;
  }

int main() {
  
  int (*res)(int, int);  
  // declaration a function pointer, of two integer parameter and 
  an int return type.
  
  res = &add;  
  // assigning address of function to pointer.
  
  int fin = res(5, 6); 
  // function call, through pointer.

  std::cout<<fin<<std::endl;
  return 0;
}
